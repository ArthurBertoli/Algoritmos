#include <stdio.h>
#include <math.h>

int main(){

    int vetor[10], a;

    for (int i=0; i < 10; i++){
        
        printf("Digite o número do vetor: ", i);
        scanf("%d", &vetor[i]);
    }

    for(int x = 0; x<10; x++){
        for(int y = x; y < 10; y++){
            if(vetor[x] > vetor[y]){
                a = vetor[x];
                vetor[x] = vetor[y];
                vetor[y] =  a;
            }
        }
    }
    for(int i = 0; i < 10; i++){
        printf("Número %d é: %d \n", i, vetor[i]);
    }
    return 0;
}
